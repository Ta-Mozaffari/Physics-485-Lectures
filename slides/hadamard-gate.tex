\begin{frame}{Hadamard gate: basis change}
  \begin{columns}[c]
    \column{0.60\textwidth}
    \[
      H=\frac{1}{\sqrt2}\begin{pmatrix}1&1\\1&-1\end{pmatrix},\qquad
      H\begin{pmatrix}\alpha\\\beta\end{pmatrix}
       =\frac{1}{\sqrt2}\begin{pmatrix}\alpha+\beta\\\alpha-\beta\end{pmatrix}.
    \]
    \[
      H|0\rangle=\frac{1}{\sqrt2}
      \begin{pmatrix}1&1\\1&-1\end{pmatrix}
      \begin{pmatrix}1\\0\end{pmatrix}
      =\frac{1}{\sqrt2}\begin{pmatrix}1\\1\end{pmatrix}=|+\rangle.
    \]
    On the Bloch sphere: a $180^\circ$ rotation about
    $(\hat x+\hat z)/\sqrt2$, giving $(x,y,z)\mapsto(z,-y,x)$.
    \column{0.34\textwidth}
    \centering\blochgate{0,1.45}{$|0\rangle$}{1.45,0}{$|+\rangle$}
    \small Blue: input \quad Red: output
  \end{columns}
\end{frame}
